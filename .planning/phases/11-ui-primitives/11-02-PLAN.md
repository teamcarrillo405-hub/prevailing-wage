  <files>src/client/components/ui/PageHeader.tsx</files>
  <action>
    Create `src/client/components/ui/PageHeader.tsx`:

    ```typescript
    import React from 'react';
    import { cn } from '../../lib/utils';

    interface PageHeaderProps {
      title: string;
      subtitle?: string;
      action?: React.ReactNode;
      className?: string;
    }

    export function PageHeader({ title, subtitle, action, className }: PageHeaderProps) {
      return (
        <div className={cn('flex items-center justify-between mb-6', className)}>
          <div>
            <h1 className="font-headline text-2xl text-text-primary">{title}</h1>
            {subtitle && (
              <p className="text-sm text-text-secondary mt-1">{subtitle}</p>
            )}
          </div>
          {action && (
            <div className="ml-4 flex-shrink-0">{action}</div>
          )}
        </div>
      );
    }
    ```

    Key decisions:
    - Uses `mb-6` (not mb-8) to match planning context spec. The mb-8 in RESEARCH.md was the existing inline pattern; the spec says mb-6.
    - Action slot is conditionally rendered — if action is undefined, no empty div is rendered.
    - h1 uses font-headline (maps to Oswald via Phase 10 @layer base) and text-text-primary token.
    - className prop merges via cn() for caller overrides (Phase 12 may need different margin on specific pages).
  </action>

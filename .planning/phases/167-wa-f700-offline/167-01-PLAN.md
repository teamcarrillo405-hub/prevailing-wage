### Task 2: Offline queue library
Create src/client/lib/offlineQueue.ts:
```typescript
const QUEUE_KEY = 'hcc_offline_queue';

export interface OfflineEvent {
  id: string;
  type: 'clock_in' | 'clock_out';
  payload: Record<string, unknown>;
  timestamp: string;
  retries: number;
}

export function enqueueEvent(type: OfflineEvent['type'], payload: Record<string, unknown>): void {
  const queue = getQueue();
  queue.push({ id: crypto.randomUUID(), type, payload, timestamp: new Date().toISOString(), retries: 0 });
  localStorage.setItem(QUEUE_KEY, JSON.stringify(queue));
}

export function getQueue(): OfflineEvent[] {
  try { return JSON.parse(localStorage.getItem(QUEUE_KEY) ?? '[]'); } catch { return []; }
}

export function removeFromQueue(id: string): void {
  const queue = getQueue().filter(e => e.id !== id);
  localStorage.setItem(QUEUE_KEY, JSON.stringify(queue));
}

export async function flushQueue(apiPost: (url: string, data: unknown) => Promise<unknown>): Promise<{ sent: number; conflicts: string[] }> {
  const queue = getQueue();
  const conflicts: string[] = [];
  let sent = 0;
  for (const event of queue) {
    try {
      await apiPost(`/clock-entries/${event.type.replace('_', '-')}`, event.payload);
      removeFromQueue(event.id);
      sent++;
    } catch (err: any) {
      if (err?.response?.status === 409) {
        conflicts.push(`${event.type} at ${event.timestamp} — conflict with existing entry`);
        removeFromQueue(event.id);
      }
    }
  }
  return { sent, conflicts };
}
```

      },
    });

    **POST /preview route:**
    - Wrap multer in manual invocation to catch MulterError (per research Pitfall 4):
      ```typescript
      importRouter.post('/preview', (req, res, next) => {
        upload.single('file')(req, res, async (err) => {
          if (err instanceof multer.MulterError) {
            return res.status(400).json({ error: err.code === 'LIMIT_FILE_SIZE' ? 'File too large. Maximum 5 MB.' : err.message });
          }
          if (err) return res.status(400).json({ error: err.message });
          // ... route body
        });
      });
      ```
    - Require `weekId` from `req.body.weekId` (multipart form field alongside the file).
    - Check `req.file` exists, return 400 "No file uploaded" if not.
    - Look up week via `getPayrollWeek(weekId)`. 404 if not found.
    - Auth: `assertProjectAccess(db, week.projectId, req.user!.userId)`. Catch => status/message.
    - Submitted check: `if (week.submittedAt) return res.status(423).json({ error: 'This payroll week is submitted and cannot be modified.' })` (per D-10).
    - Call `parseImportFile(req.file.buffer, weekId, week.projectId, db)`.
    - If parseImportFile throws (unknown provider), return 400 with the error message.
    - Return 200 with the ImportPreviewResult JSON.

    **POST /commit route:**
    - JSON body (no multer): `{ weekId: string, provider: 'quickbooks' | 'adp', matched: ImportedRow[], unmatchedCount: number, sourceFilename?: string }`.
    - Auth + assertProjectAccess + submittedAt check — same pattern as preview.
    - Re-validate: check that none of the matched workers already have entries for this week (conflict check). If any conflict found, return 409 with details.
    - Insert payrollEntries in a loop using `db.insert(payrollEntries).values(...)` for each matched row:
      ```typescript
      const now = new Date().toISOString();
      for (const row of body.matched) {

          monOt: row.monOt, tueOt: row.tueOt, wedOt: row.wedOt, thuOt: row.thuOt, friOt: row.friOt, satOt: row.satOt, sunOt: row.sunOt,
          monDt: 0, tueDt: 0, wedDt: 0, thuDt: 0, friDt: 0, satDt: 0, sunDt: 0,
          baseRateSnapshot: row.baseRateSnapshot,
          fringeRateSnapshot: row.fringeRateSnapshot,
          grossWages: null, deductions: 0, netPay: null,
          fringeHealthWelfare: null, fringePension: null, fringeVacation: null, fringeTraining: null,

        });
      }
      ```

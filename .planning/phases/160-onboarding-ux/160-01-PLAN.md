### Task 3: DueSoonPanel urgency colors
Read DueSoonPanel. Find where items are rendered. Add urgency coloring based on days until due:

```typescript
function getUrgency(dueDateStr: string): 'error' | 'warning' | 'muted' {
  const daysUntil = Math.ceil((new Date(dueDateStr).getTime() - Date.now()) / (1000 * 60 * 60 * 24));
  if (daysUntil <= 1) return 'error';
  if (daysUntil <= 3) return 'warning';
  return 'muted';
}

function getRelativeLabel(dueDateStr: string): string {
  const daysUntil = Math.ceil((new Date(dueDateStr).getTime() - Date.now()) / (1000 * 60 * 60 * 24));
  if (daysUntil <= 0) return 'Overdue';
  if (daysUntil === 1) return 'Due tomorrow';
  return `Due in ${daysUntil} days`;
}
```

Wrap each item with urgency class:
```tsx
const urgency = getUrgency(item.dueDate);
<div className={`flex items-center justify-between py-2 px-3 rounded-lg ${
  urgency === 'error' ? 'bg-red-500/10 border border-red-500/20' :
  urgency === 'warning' ? 'bg-amber-500/10 border border-amber-500/20' :
  'bg-surface-card'
}`}>
  <span className="text-sm">{item.projectName}</span>
  <span className={`text-xs font-medium ${urgency === 'error' ? 'text-red-400' : urgency === 'warning' ? 'text-amber-400' : 'text-surface-muted'}`}>
    {getRelativeLabel(item.dueDate)}
  </span>
</div>
```

Commit: `feat(due-soon): urgency color coding with relative time labels`

# Plan 168-01: Security Hardening

## Objective
Three hardening measures: (1) sameSite:'strict' on session cookie, (2) SSN API response audit — mask SSNs to last-4 in all API responses, (3) rate limiting on /api/auth/login to prevent brute force.

### Task 3: Rate limiting on auth endpoints
Create src/server/middleware/rateLimiter.ts:
```typescript
import { Request, Response, NextFunction } from 'express';

const attempts = new Map<string, { count: number; resetAt: number }>();
const WINDOW_MS = 15 * 60 * 1000; // 15 min
const MAX_ATTEMPTS = 10;

export function authRateLimiter(req: Request, res: Response, next: NextFunction) {
  const key = req.ip ?? 'unknown';
  const now = Date.now();
  const record = attempts.get(key);

  if (record && now < record.resetAt) {
    if (record.count >= MAX_ATTEMPTS) {
      return res.status(429).json({
        error: 'TOO_MANY_REQUESTS',
        message: `Too many login attempts. Try again in ${Math.ceil((record.resetAt - now) / 60000)} minutes.`,
        retryAfter: Math.ceil((record.resetAt - now) / 1000)
      });
    }
    record.count++;
  } else {
    attempts.set(key, { count: 1, resetAt: now + WINDOW_MS });
  }
  next();
}

export function resetAuthRateLimit(ip: string) {
  attempts.delete(ip);
}
```

Apply to login route in src/server/routes/auth.ts:
```typescript
import { authRateLimiter, resetAuthRateLimit } from '../middleware/rateLimiter';
// On login route:
router.post('/login', authRateLimiter, async (req, res) => {
  // ... existing logic ...
  // On success: resetAuthRateLimit(req.ip ?? '');
});
```

Commit: `feat(security): in-memory rate limiter on /api/auth/login (10 req/15min)`

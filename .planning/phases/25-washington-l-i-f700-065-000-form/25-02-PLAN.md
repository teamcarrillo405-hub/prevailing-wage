   ```typescript
   //   [Copy results from tests/services/f700.test.ts header comment]
   //

   //   - No DT (double-time) column — WA OT is weekly 1.5x, no daily DT threshold (RCW 49.28.010)

   // Pattern: baseY = first worker ST row; blockSpacing = ST-to-next-ST; subRowSpacing = ST-to-OT
   function getWorkerRowLY(workerIdx: number): { st: number; ot: number } {

   // ── Main export: fillF700 ─────────────────────────────────────────────────

   export async function fillF700(
     data: F700Data,
     templateBytes: Uint8Array | Buffer,
   ): Promise<Uint8Array> {
     const pdfDoc = await PDFDocument.load(templateBytes);
     const font = await pdfDoc.embedFont(StandardFonts.Helvetica);
     const boldFont = await pdfDoc.embedFont(StandardFonts.HelveticaBold);

     const TEXT_SIZE = 7;
     const SMALL_SIZE = 6;
     const black = rgb(0, 0, 0);

     // Chunk workers into groups of ROWS_PER_PAGE
     const chunks: F700WorkerRow[][] = [];
     for (let i = 0; i < data.workers.length; i += ROWS_PER_PAGE) {
       chunks.push(data.workers.slice(i, i + ROWS_PER_PAGE));
     }
     const totalPageSets = chunks.length === 0 ? 1 : chunks.length;

     // Copy additional template pages for overflow (mirrors a1131Generator pattern)
     // TODO: determine if WA form has 1 or 2 pages. If 2 (worker page + cert page),
     // use setIdx * 2 indexing (same as A-1-131). If 1 page, use setIdx directly.
     // Update based on PDF inspection (page count from 25-01-01).
     const PAGE_COUNT_PER_SET = 1; // TBD — update after PDF inspection
     for (let setIdx = 1; setIdx < totalPageSets; setIdx++) {
       for (let p = 0; p < PAGE_COUNT_PER_SET; p++) {
         const [extraPage] = await pdfDoc.copyPages(pdfDoc, [p]);
         pdfDoc.addPage(extraPage);
       }
     }

     const allPages = pdfDoc.getPages();

     // Fill each worker-grid page
     for (let setIdx = 0; setIdx < totalPageSets; setIdx++) {
       const workerPage = allPages[setIdx * PAGE_COUNT_PER_SET];

       if (HAS_ROTATION) {
         // Apply CTM to draw in landscape space (same as A-1-131 pattern)
         workerPage.pushOperators(
           pushGraphicsState(),
           concatTransformationMatrix(0, 1, -1, 0, PAGE_WIDTH, 0),
         );
       }

       // Helper: draw text at position (landscape if rotated, portrait otherwise)
       const dt = (
         text: string,
         lx: number,
         ly: number,
         opts?: { size?: number; bold?: boolean; maxWidth?: number },
       ) => {
         workerPage.drawText(text, {
           x: lx,
           y: ly,
           size: opts?.size ?? TEXT_SIZE,
           font: opts?.bold ? boldFont : font,
           color: black,
           maxWidth: opts?.maxWidth,
         });
       };

       // Header fields
       dt(data.contractorName,    HEADER.contractorName.lx,    HEADER.contractorName.ly,    { maxWidth: 340 });
       dt(data.ubiNumber,         HEADER.ubiNumber.lx,         HEADER.ubiNumber.ly);
       dt(data.contractorAddress, HEADER.address.lx,           HEADER.address.ly,           { maxWidth: 200 });
       dt(data.lniCertificate,    HEADER.lniCertificate.lx,    HEADER.lniCertificate.ly,    { maxWidth: 200 });
       dt(data.wcAccount,         HEADER.wcAccount.lx,         HEADER.wcAccount.ly,         { maxWidth: 200 });
       dt(data.payrollNumber,     HEADER.payrollNo.lx,         HEADER.payrollNo.ly);
       dt(data.weekEndingDate,    HEADER.weekEndingDate.lx,    HEADER.weekEndingDate.ly);
       dt(data.projectName,       HEADER.projectName.lx,       HEADER.projectName.ly,       { maxWidth: 280 });
       dt(data.projectLocation,   HEADER.projectLocation.lx,  HEADER.projectLocation.ly,   { maxWidth: 200 });
       dt(data.contractNo,        HEADER.contractNo.lx,        HEADER.contractNo.ly,        { maxWidth: 200 });

       if (totalPageSets > 1) {
         dt(`Page ${setIdx + 1} of ${totalPageSets}`, 940, 556);
       }

       // Worker rows
       const chunk = chunks[setIdx] ?? [];
       for (let i = 0; i < chunk.length; i++) {
         const w = chunk[i];
         const rowY = getWorkerRowLY(i);
         const stY = rowY.st;
         const otY = rowY.ot;

         // Entry number
         dt(String(w.entryNo), COL.entryNo, stY, { size: SMALL_SIZE });

         // Worker name (Last, First format)
         let displayName = w.workerName;
         if (w.workerName.includes(',')) {
           const [last, rest] = w.workerName.split(',');
           displayName = `${(last ?? '').trim()}, ${(rest ?? '').trim()}`;
         }
         dt(displayName, COL.workerName, stY, { size: SMALL_SIZE, maxWidth: 90 });

         // Identifying No
         dt(w.identifyingNo, COL.identifyingNo, stY, { size: SMALL_SIZE });

         // Labor type: J or RA
         const laborLabel = w.laborType === 'journeyworker' ? 'J' : 'RA';
         dt(laborLabel, COL.laborType, stY, { size: SMALL_SIZE, bold: true });

         // WA Trade Code (4-letter)
         dt(w.waTradeCode, COL.waTradeCode, stY, { size: SMALL_SIZE });

         // Classification description
         dt(w.classification, COL.classification, stY, { size: SMALL_SIZE, maxWidth: 55 });

         // ST hours (Mon–Sun — WA form day order)
         dt(fmtHours(w.monSt), COL.monHours, stY, { size: SMALL_SIZE });
         dt(fmtHours(w.tueSt), COL.tueHours, stY, { size: SMALL_SIZE });
         dt(fmtHours(w.wedSt), COL.wedHours, stY, { size: SMALL_SIZE });
         dt(fmtHours(w.thuSt), COL.thuHours, stY, { size: SMALL_SIZE });
         dt(fmtHours(w.friSt), COL.friHours, stY, { size: SMALL_SIZE });
         dt(fmtHours(w.satSt), COL.satHours, stY, { size: SMALL_SIZE });
         dt(fmtHours(w.sunSt), COL.sunHours, stY, { size: SMALL_SIZE });
         dt(fmtHours(w.totalSt), COL.totalHours, stY, { size: SMALL_SIZE });
         dt(fmtDollar(w.baseRate), COL.baseRate, stY, { size: SMALL_SIZE });

         // OT row (WA: 1.5x after 40 hrs/week; no DT row)
         dt(fmtHours(w.monOt), COL.monHours, otY, { size: SMALL_SIZE });
         dt(fmtHours(w.tueOt), COL.tueHours, otY, { size: SMALL_SIZE });
         dt(fmtHours(w.wedOt), COL.wedHours, otY, { size: SMALL_SIZE });
         dt(fmtHours(w.thuOt), COL.thuHours, otY, { size: SMALL_SIZE });
         dt(fmtHours(w.friOt), COL.friHours, otY, { size: SMALL_SIZE });
         dt(fmtHours(w.satOt), COL.satHours, otY, { size: SMALL_SIZE });
         dt(fmtHours(w.sunOt), COL.sunHours, otY, { size: SMALL_SIZE });
         dt(fmtHours(w.totalOt), COL.totalHours, otY, { size: SMALL_SIZE });
         dt(fmtDollar(w.otRate), COL.baseRate, otY, { size: SMALL_SIZE });

         // Pay summary — on ST row
         dt(fmtDollar(w.grossWages),   COL.grossWages,   stY, { size: SMALL_SIZE });
         dt(fmtDollar(w.deductions),   COL.deductions,   stY, { size: SMALL_SIZE });
         dt(fmtDollar(w.netPay),       COL.netPay,       stY, { size: SMALL_SIZE });
         dt(fmtDollar(w.fringeCredit), COL.fringeCredit, otY, { size: SMALL_SIZE });
       }

       if (HAS_ROTATION) {
         workerPage.pushOperators(popGraphicsState());
       }
     }

     // Certification page — fill if the WA form has a cert section
     // TODO: if WA form has a second page (cert), add filling here.
     // If WA form is single-page (worker grid only), omit this section.
     // Determine from PDF inspection page count (25-01-01).

     return pdfDoc.save();
   }
   ```

   ```typescript
   // ── GET /api/export/f700/:weekId ──────────────────────────────────────────
   // Washington L&I F700-065-000 — state-gated to WA projects only

   router.get('/f700/:weekId', async (req, res) => {
     const weekId = req.params.weekId as string;
     const userId = req.user!.userId;

     // 1. Load payroll week
     const week = await getPayrollWeek(weekId);
     if (!week) {
       res.status(404).json({ error: 'Payroll week not found' });
       return;
     }

     // 2. Verify project ownership
     const db = getDb();
     const [project] = await db
       .select()
       .from(projects)
       .where(eq(projects.id, week.projectId))
       .limit(1);

     if (!project) {
       res.status(404).json({ error: 'Project not found' });
       return;
     }
     if (project.userId !== userId) {
       res.status(403).json({ error: 'Access denied' });
       return;
     }

     // 3. State gate — F700-065-000 is WA-only
     if (project.state !== 'WA') {
       res.status(400).json({ error: 'F700-065-000 is only available for Washington projects' });
       return;
     }

     // 4. Load payroll entries
     const entries = await getPayrollEntries(weekId);

     // 5. Map entries to F700WorkerRow[]
     type EntryRow = (typeof entries)[number];
     const workerRows: F700WorkerRow[] = entries.map((row: EntryRow, index: number) => {
       const e = row.entry;
       const totalSt = e.monSt + e.tueSt + e.wedSt + e.thuSt + e.friSt + e.satSt + e.sunSt;
       const totalOt = e.monOt + e.tueOt + e.wedOt + e.thuOt + e.friOt + e.satOt + e.sunOt;
       const baseRate = e.baseRateSnapshot;

       // WA trade code: use waTradeCode override from DB if set,
       // otherwise fall back to tradeCode if it's a known WA code,
       // otherwise leave blank (user should set waTradeCode override via WorkersPage)
       const waTradeCode = row.waTradeCode
         ?? (row.tradeCode && WA_TRADE_CODES[row.tradeCode] ? row.tradeCode : '');

       return {
         entryNo: index + 1,
         workerName: row.workerName,
         identifyingNo: '',  // SSN last 4 — privacy default
         laborType: (row.laborType === 'foreman' ? 'journeyworker' : row.laborType) as 'journeyworker' | 'apprentice',
         waTradeCode,
         classification: row.tradeDescription,
         monSt: e.monSt, monOt: e.monOt,
         tueSt: e.tueSt, tueOt: e.tueOt,
         wedSt: e.wedSt, wedOt: e.wedOt,
         thuSt: e.thuSt, thuOt: e.thuOt,
         friSt: e.friSt, friOt: e.friOt,
         satSt: e.satSt, satOt: e.satOt,
         sunSt: e.sunSt, sunOt: e.sunOt,
         totalSt,
         totalOt,
         baseRate,
         otRate: baseRate * 1.5,
         grossWages: e.grossWages ?? 0,
         deductions: e.deductions,
         netPay: e.netPay ?? 0,
         fringeCredit: e.fringeRateSnapshot * (totalSt + totalOt),
       };
     });

     // 6. Build F700Data
     const f700Data: F700Data = {
       contractorName: project.name,
       contractorAddress: `${project.county}, ${project.state}`,
       ubiNumber: project.ubiNumber || '',
       lniCertificate: project.lniCertificate || '',
       wcAccount: project.wcAccount || '',
       projectName: project.name,
       projectLocation: `${project.county}, ${project.state}`,
       contractNo: project.wdIdentifier ?? '',
       weekEndingDate: formatDate(week.weekEndingDate),
       payrollNumber: week.amendmentNumber != null && week.originalWeekId != null
         ? `${week.payrollNumber} (AMENDED ${week.amendmentNumber})`
         : String(week.payrollNumber),
       workers: workerRows,
     };

     // 7. Load template + fill
     const templatePath = path.join(process.cwd(), 'assets', 'f700-official.pdf');
     const templateBytes = readFileSync(templatePath);
     const filledPdf = await fillF700(f700Data, templateBytes);

     // 8. Stream as PDF download
     const filename = week.amendmentNumber != null
       ? `f700-${week.payrollNumber}-amended-${week.amendmentNumber}.pdf`
       : `f700-${week.payrollNumber}.pdf`;
     res.setHeader('Content-Type', 'application/pdf');
     res.setHeader('Content-Disposition', `attachment; filename="${filename}"`);
     res.setHeader('Content-Length', filledPdf.length);
     res.end(Buffer.from(filledPdf));
   });
   ```

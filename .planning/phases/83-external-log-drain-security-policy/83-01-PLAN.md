    ```typescript
    import pino from 'pino';

    const isDev  = process.env.NODE_ENV !== 'production';
    const isTest = process.env.NODE_ENV === 'test';
    const token  = process.env.LOGTAIL_TOKEN;

    /**
     * Transport selection (Phase 83 SEC-07):
     *   - test           -> no transport (suppress external calls; tests already
     *                       set NODE_ENV=test before import)
     *   - dev, no token  -> pino-pretty to console (existing dev experience)
     *   - dev or prod, token present
     *                    -> @logtail/pino HTTPS drain to Better Stack
     *   - prod, no token -> no transport (raw JSON to stdout — Render captures)
     */
    function buildTransport() {
      if (isTest) return undefined;
      if (token) {
        return pino.transport({
          target: '@logtail/pino',
          options: { sourceToken: token },
        });
      }
      if (isDev) {
        return pino.transport({
          target: 'pino-pretty',
          options: { colorize: true, translateTime: 'SYS:HH:MM:ss', ignore: 'pid,hostname' },
        });
      }
      return undefined;
    }

    export const logger = pino(
      { level: process.env.LOG_LEVEL ?? (isDev ? 'debug' : 'info') },
      buildTransport(),
    );
    ```

    Key constraints (DO NOT violate):
    - The `export const logger` symbol name MUST stay identical — `pino-http` middleware in `index.ts` imports `{ logger }` by name.
    - The `level` config MUST keep `process.env.LOG_LEVEL ?? (isDev ? 'debug' : 'info')` ordering — production defaults to `info`.
    - Pass the transport stream as the SECOND argument to `pino()` — never as a property of the options object (Research Pitfall 1: TypeScript error "ThreadStream not assignable to LoggerOptions").
    - The `isTest` branch MUST return `undefined` — never construct a transport that touches the network in test env (Research Pitfall 2: CI hangs).
    - When token is present in dev, prefer drain (matches research recommendation — drain mirroring is overkill).

    Do NOT change `src/server/index.ts` `pinoHttp({ logger })` middleware — the transport flows through automatically (Research Pitfall 4: middleware uses shared logger).

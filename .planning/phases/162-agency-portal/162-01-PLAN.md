# Plan 162-01: Agency Portal — Read-Only Reviewer Login + Review Stamp

## Objective
Add an `agency_reviewer` role to the users table. Reviewers can log in and see a read-only view of payroll weeks for projects they're granted access to. They can leave a review comment/stamp. No contractor data mutation is allowed.

## SPARC

### Specification
- `users.role` gets new value `'agency_reviewer'` (existing values: `'owner'`, `'admin'`, `'member'`)
- New table `reviewer_project_access (reviewer_user_id, project_id, granted_by, granted_at)`
- New table `payroll_week_comments (id, week_id, user_id, comment, review_stamp, created_at)`
- GET /api/agency/projects — returns projects the reviewer has access to (joins reviewer_project_access)
- GET /api/agency/projects/:id/weeks — returns payroll weeks (read-only)
- POST /api/agency/weeks/:weekId/review — creates a comment/stamp
- Client route `/agency` → AgencyReviewPage (read-only, role-gated)

### Task 2: Agency routes
Create src/server/routes/agency.ts:
```typescript
import { Router } from 'express';
import { db } from '../db';

const router = Router();

// Middleware: only agency_reviewer role
router.use((req: any, res, next) => {
  if (!req.session?.userId) return res.status(401).json({ error: 'UNAUTHENTICATED' });
  // role check done at query level via reviewer_project_access join
  next();
});

router.get('/projects', async (req: any, res) => {
  const rows = await db.all(`
    SELECT p.id, p.name, p.status, p.funding_type, rpa.granted_at
    FROM projects p
    JOIN reviewer_project_access rpa ON rpa.project_id = p.id
    WHERE rpa.reviewer_user_id = ?
  `, [req.session.userId]);
  res.json(rows);
});

router.get('/projects/:id/weeks', async (req: any, res) => {
  const access = await db.get(`SELECT 1 FROM reviewer_project_access WHERE reviewer_user_id = ? AND project_id = ?`, [req.session.userId, req.params.id]);
  if (!access) return res.status(403).json({ error: 'NO_ACCESS' });
  const weeks = await db.all(`SELECT * FROM payroll_weeks WHERE project_id = ? ORDER BY week_ending_date DESC`, [req.params.id]);
  res.json(weeks);
});

router.post('/weeks/:weekId/review', async (req: any, res) => {
  const { comment, review_stamp } = req.body;
  if (!comment) return res.status(400).json({ error: 'COMMENT_REQUIRED' });
  const result = await db.run(
    `INSERT INTO payroll_week_comments (week_id, user_id, comment, review_stamp) VALUES (?, ?, ?, ?)`,
    [req.params.weekId, req.session.userId, comment, review_stamp ?? 'pending']
  );
  res.status(201).json({ id: result.lastID });
});

export default router;
```

Note: agency.ts uses raw db.all/db.get/db.run (SQLite style) — if the project uses drizzle, adapt to drizzle syntax by reading src/server/db/schema.ts first.

### Task 3: AgencyReviewPage
Create src/client/pages/AgencyReviewPage.tsx:
```tsx
import { useState } from 'react';
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import api from '../lib/api';
import { PageHeader } from '../components/ui/PageHeader';
import { Badge } from '../components/ui/Badge';

export function AgencyReviewPage() {
  const [selectedProject, setSelectedProject] = useState<number | null>(null);
  const [comment, setComment] = useState('');
  const [stamp, setStamp] = useState<'approved' | 'flagged' | 'pending'>('pending');
  const [selectedWeek, setSelectedWeek] = useState<number | null>(null);
  const qc = useQueryClient();

  const { data: projects = [] } = useQuery({ queryKey: ['agency-projects'], queryFn: () => api.get('/agency/projects').then(r => r.data) });
  const { data: weeks = [] } = useQuery({ queryKey: ['agency-weeks', selectedProject], queryFn: () => selectedProject ? api.get(`/agency/projects/${selectedProject}/weeks`).then(r => r.data) : [], enabled: !!selectedProject });

  const submitReview = useMutation({
    mutationFn: () => api.post(`/agency/weeks/${selectedWeek}/review`, { comment, review_stamp: stamp }),
    onSuccess: () => { setComment(''); qc.invalidateQueries({ queryKey: ['agency-weeks'] }); }
  });

  return (
    <div className="max-w-4xl mx-auto py-6 px-4">
      <PageHeader title="Agency Review Portal" subtitle="Read-only access to assigned projects" />
      <div className="grid grid-cols-1 md:grid-cols-3 gap-6 mt-6">
        <div className="space-y-2">
          <p className="text-xs text-surface-muted font-medium uppercase tracking-wide mb-3">Projects</p>
          {projects.map((p: any) => (
            <button key={p.id} onClick={() => setSelectedProject(p.id)}
              className={`w-full text-left px-3 py-2 rounded-lg text-sm ${selectedProject === p.id ? 'bg-brand-gold text-black font-medium' : 'bg-surface-card text-white hover:bg-surface-card/80'}`}>
              {p.name}
            </button>
          ))}
          {projects.length === 0 && <p className="text-sm text-surface-muted">No projects assigned.</p>}
        </div>
        <div className="md:col-span-2 space-y-3">
          {weeks.map((w: any) => (
            <div key={w.id} className={`bg-surface-card rounded-xl p-4 border ${selectedWeek === w.id ? 'border-brand-gold' : 'border-surface-card'}`} onClick={() => setSelectedWeek(w.id)}>
              <div className="flex items-center justify-between">
                <p className="text-sm font-medium text-white">Week ending {w.week_ending_date}</p>
                <Badge variant={w.submitted_at ? 'compliant' : 'neutral'}>{w.submitted_at ? 'Submitted' : 'Not submitted'}</Badge>
              </div>
            </div>
          ))}
          {selectedWeek && (
            <div className="bg-surface-card rounded-xl p-4 mt-4">
              <p className="text-sm font-medium text-white mb-3">Leave Review</p>
              <select value={stamp} onChange={e => setStamp(e.target.value as any)} className="w-full bg-surface-page text-white text-sm rounded px-3 py-2 mb-3 border border-surface-card">
                <option value="pending">Pending</option>
                <option value="approved">Approved</option>
                <option value="flagged">Flagged</option>
              </select>
              <textarea value={comment} onChange={e => setComment(e.target.value)} placeholder="Review comments..." rows={3} className="w-full bg-surface-page text-white text-sm rounded px-3 py-2 mb-3 border border-surface-card outline-none focus:border-brand-gold resize-none" />
              <button onClick={() => submitReview.mutate()} disabled={!comment} className="px-4 py-2 bg-brand-gold text-black text-sm font-medium rounded hover:opacity-90 disabled:opacity-50">Submit Review</button>
            </div>
          )}
        </div>
      </div>
    </div>
  );
}
```

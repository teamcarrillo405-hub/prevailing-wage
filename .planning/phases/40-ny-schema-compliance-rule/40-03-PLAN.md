       Use `getDb()` inline — do NOT rename the `_db` parameter (Pitfall 2 from research). Import `eq` if not already imported.

    2. Determine if project is NY:

    3. Inside the per-entry loop (where violations are computed), add NY daily OT check. This should run BEFORE or INSTEAD OF the existing weekly CWHSSA check for NY projects:
       ```typescript
       if (isNY) {
         const dayNames = ['Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat', 'Sun'];
         const days = [
           { st: entry.monSt, ot: entry.monOt },
           { st: entry.tueSt, ot: entry.tueOt },
           { st: entry.wedSt, ot: entry.wedOt },
           { st: entry.thuSt, ot: entry.thuOt },
           { st: entry.friSt, ot: entry.friOt },
           { st: entry.satSt, ot: entry.satOt },
           { st: entry.sunSt, ot: entry.sunOt },
         ];
         for (let i = 0; i < days.length; i++) {
           const total = (days[i].st ?? 0) + (days[i].ot ?? 0);
           if (total > 8) {
             violations.push({
               // ... match existing ComplianceViolation shape
               violationType: 'cwhssa-ot',
               message: `NY daily OT: ${dayNames[i]} hours (${total}) exceed 8-hour daily threshold`,
               // ... include workerId, workerName, entryId as per existing pattern
             });
           }
         }
       }
       ```

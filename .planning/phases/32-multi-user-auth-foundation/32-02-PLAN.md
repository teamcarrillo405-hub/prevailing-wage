```typescript
// src/server/utils/assertProjectAccess.ts
export async function assertProjectAccess(
  db: DrizzleDb,
  projectId: string,
  userId: string,
): Promise<Project>
// Throws { status: 404, message: 'Project not found' } if project does not exist
// Throws { status: 403, message: 'Access denied' } if user is not a member
// Returns full Project row on success — callers can use it directly without a second SELECT
```

<!-- Standard import line for all route files: -->
```typescript

```

<!-- Standard call site replacement pattern: -->
```typescript
// BEFORE (inline pattern):
const [project] = await db.select().from(projects).where(eq(projects.id, projectId)).limit(1);
if (!project) { res.status(404).json({ error: 'Project not found' }); return; }
if (project.userId !== userId) { res.status(403).json({ error: 'Access denied' }); return; }

// AFTER:
let project: Project;
try {
  project = await assertProjectAccess(db, projectId, userId);
} catch (err: any) {
  res.status(err.status ?? 500).json({ error: err.message ?? 'Internal server error' });
  return;
}
// project is now the full Project row — use as before
```

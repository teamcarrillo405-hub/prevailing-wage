```typescript
import { describe, it, expect, beforeAll } from 'vitest';
import supertest from 'supertest';
import { app } from '../../src/server/app.js';  // adjust import path to match project

describe('Cross-tenant IDOR protection', () => {
  let cookieA: string;
  let cookieB: string;
  let projectIdA: string;
  let projectIdB: string;

  beforeAll(async () => {
    // Register userA and userB via HTTP (same as all other route tests)
    process.env.JWT_SECRET = 'test-secret-at-least-32-characters-long-xx';
    process.env.NODE_ENV = 'test';

    cookieA = await registerAndLogin('cross-a');
    cookieB = await registerAndLogin('cross-b');

    // Create one project per user — POST /projects now inserts project_members row
    projectIdA = await createProject(cookieA, 'Project Alpha');
    projectIdB = await createProject(cookieB, 'Project Beta');
  });

  // projects.ts (3 routes)
  it('GET /api/projects/:id — userB gets 403 on userA project', async () => {
    const res = await supertest(app).get(`/api/projects/${projectIdA}`).set('Cookie', cookieB);
    expect(res.status).toBe(403);
  });

  it('PATCH /api/projects/:id — userB gets 403 on userA project', async () => {
    const res = await supertest(app).patch(`/api/projects/${projectIdA}`)
      .set('Cookie', cookieB).send({ name: 'Hacked' });
    expect(res.status).toBe(403);
  });

  it('DELETE /api/projects/:id — userB gets 403 on userA project', async () => {
    // Use a throwaway project so userA's main project remains for other tests
    const throwawayId = await createProject(cookieA, 'Throwaway');
    const res = await supertest(app).delete(`/api/projects/${throwawayId}`).set('Cookie', cookieB);
    expect(res.status).toBe(403);
  });

  // workers.ts (2 representative routes)
  it('GET /api/workers/:projectId/workers — userB gets 403 on userA project', async () => {
    const res = await supertest(app).get(`/api/workers/${projectIdA}/workers`).set('Cookie', cookieB);
    expect(res.status).toBe(403);
  });

  it('POST /api/workers/:projectId/workers — userB gets 403 on userA project', async () => {
    const res = await supertest(app).post(`/api/workers/${projectIdA}/workers`)
      .set('Cookie', cookieB).send({ name: 'Ghost Worker', trade: 'Laborer' });
    expect(res.status).toBe(403);
  });

  // reports.ts (1 route)
  it('GET /api/reports/:projectId/fringe-summary — userB gets 403 on userA project', async () => {
    const res = await supertest(app).get(`/api/reports/${projectIdA}/fringe-summary`).set('Cookie', cookieB);
    expect(res.status).toBe(403);
  });

  // compliance.ts (1 route)
  it('GET /api/compliance/project/:projectId — userB gets 403 on userA project', async () => {
    const res = await supertest(app).get(`/api/compliance/project/${projectIdA}`).set('Cookie', cookieB);
    expect(res.status).toBe(403);
  });

  // export.ts (1 route — week-scoped, uses assertProjectAccess via payroll week lookup)
  it('GET /api/export/wh347/:weekId — userB gets 403 on userA payroll week', async () => {
    // weekIdA is a payroll week belonging to projectIdA; see beforeAll setup
    const res = await supertest(app).get(`/api/export/wh347/${weekIdA}`).set('Cookie', cookieB);
    expect(res.status).toBe(403);
  });

  // payroll.ts (1 route — week-scoped)
  it('GET /api/payroll/:projectId/weeks — userB gets 403 on userA project payroll weeks', async () => {
    const res = await supertest(app).get(`/api/payroll/${projectIdA}/weeks`).set('Cookie', cookieB);
    expect(res.status).toBe(403);
  });

  // Symmetric: userA cannot access userB's project
  it('GET /api/projects/:id — userA gets 403 on userB project (symmetric)', async () => {
    const res = await supertest(app).get(`/api/projects/${projectIdB}`).set('Cookie', cookieA);
    expect(res.status).toBe(403);
  });

  // Positive control: userA can still access their own project
  it('GET /api/projects/:id — userA gets 200 on their own project', async () => {
    const res = await supertest(app).get(`/api/projects/${projectIdA}`).set('Cookie', cookieA);
    expect(res.status).toBe(200);
  });
});
```

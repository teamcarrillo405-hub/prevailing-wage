### Task 2: Password strength meter in RegisterForm
Add after the password input field:

```tsx
// Scoring function (add outside component)
function scorePassword(pw: string): number {
  let score = 0;
  if (pw.length >= 8) score++;
  if (/[A-Z]/.test(pw)) score++;
  if (/[0-9]/.test(pw)) score++;
  if (/[^A-Za-z0-9]/.test(pw)) score++;
  if (pw.length >= 12) score++;
  return score; // 0-5
}

const STRENGTH_LABELS = ['', 'Weak', 'Fair', 'Good', 'Strong', 'Very Strong'];
const STRENGTH_COLORS = ['bg-surface-muted', 'bg-red-500', 'bg-orange-500', 'bg-amber-500', 'bg-yellow-400', 'bg-green-500'];

// In component, derive from password state:
const strength = scorePassword(password);

// Render after password input:
{password.length > 0 && (
  <div className="mt-2 space-y-2">
    <div className="flex gap-1">
      {[1,2,3,4,5].map(i => (
        <div key={i} className={`h-1 flex-1 rounded-full transition-colors ${i <= strength ? STRENGTH_COLORS[strength] : 'bg-surface-muted'}`} />
      ))}
    </div>
    {strength > 0 && <p className={`text-xs ${strength >= 4 ? 'text-green-400' : strength >= 2 ? 'text-amber-400' : 'text-red-400'}`}>{STRENGTH_LABELS[strength]}</p>}
    <div className="space-y-1 text-xs">
      {[
        { label: '8+ characters', met: password.length >= 8 },
        { label: 'Uppercase letter', met: /[A-Z]/.test(password) },
        { label: 'Number or symbol', met: /[0-9^A-Za-z]/.test(password) || /[^A-Za-z0-9]/.test(password) },
      ].map(r => (
        <div key={r.label} className={`flex items-center gap-1.5 ${r.met ? 'text-green-400' : 'text-surface-muted'}`}>
          <span>{r.met ? '✓' : '○'}</span><span>{r.label}</span>
        </div>
      ))}
    </div>
  </div>
)}
```

Commit: `feat(auth): password strength meter with 5-segment bar and requirements checklist`

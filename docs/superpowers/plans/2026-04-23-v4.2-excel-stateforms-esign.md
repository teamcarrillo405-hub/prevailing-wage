# v4.2 — Excel Export + IL/NY State Forms + E-Signature Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add Excel (.xlsx) export for payroll data, Illinois IDOL CTP and New York DOL App 15 certified payroll forms, and a digital certification signature workflow on WH-347.

**Architecture:** Excel export uses `exceljs` (new dep). IL/NY forms follow the existing coordinate-based pdf-lib generator pattern from CA/WA — no new PDF template files needed since both forms are generate-from-scratch (not fillable). E-signature stores a base64 drawn signature or typed name + timestamp in a new `payroll_certifications` table; the WH-347 generator reads it and overlays it.

**Tech Stack:** `exceljs` (new), `pdf-lib` (existing), Drizzle + SQLite migrations.

**Install before starting:**
```bash
npm install exceljs
npm install -D @types/exceljs
```

### Task 2 — IL CTP PDF generator

The Illinois IDOL CTP is a text-layout certified payroll report. We generate it via pdf-lib using coordinate overlays on a blank A4 page — same pattern as the compliance summary PDF generator.

- [ ] Create `src/server/services/ilCtpGenerator.ts`:

```ts
import { PDFDocument, StandardFonts, rgb } from 'pdf-lib';
import { getDb } from '../db/index.js';
import { payrollWeeks, payrollEntries, workers, wdClassifications, projects } from '../db/schema.js';
import { eq } from 'drizzle-orm';

const MARGIN = 36;

export async function generateIlCtp(weekId: string): Promise<Uint8Array> {
  const db = getDb();
  const [weekRow] = await db.select({ week: payrollWeeks, project: projects })
    .from(payrollWeeks)
    .innerJoin(projects, eq(payrollWeeks.projectId, projects.id))
    .where(eq(payrollWeeks.id, weekId)).limit(1);
  if (!weekRow) throw new Error('Week not found');

  const entries = await db
    .select({
      workerName: workers.name,
      trade: wdClassifications.description,
      baseRate: payrollEntries.baseRateSnapshot,
      grossWages: payrollEntries.grossWages,
      deductions: payrollEntries.deductions,
      netPay: payrollEntries.netPay,
      monSt: payrollEntries.monSt, tueSt: payrollEntries.tueSt,
      wedSt: payrollEntries.wedSt, thuSt: payrollEntries.thuSt,
      friSt: payrollEntries.friSt, satSt: payrollEntries.satSt, sunSt: payrollEntries.sunSt,
      monOt: payrollEntries.monOt, tueOt: payrollEntries.tueOt,
      wedOt: payrollEntries.wedOt, thuOt: payrollEntries.thuOt,
      friOt: payrollEntries.friOt, satOt: payrollEntries.satOt, sunOt: payrollEntries.sunOt,
    })
    .from(payrollEntries)
    .innerJoin(workers, eq(payrollEntries.workerId, workers.id))
    .leftJoin(wdClassifications, eq(payrollEntries.classificationId, wdClassifications.id))
    .where(eq(payrollEntries.payrollWeekId, weekId));

  const pdfDoc = await PDFDocument.create();
  const font = await pdfDoc.embedFont(StandardFonts.Helvetica);
  const bold = await pdfDoc.embedFont(StandardFonts.HelveticaBold);
  const black = rgb(0, 0, 0);

  // Page 1: header + entries (paginate at ~8 workers per page)
  const ROWS_PER_PAGE = 8;
  const pages = Math.ceil(Math.max(entries.length, 1) / ROWS_PER_PAGE);

  for (let p = 0; p < pages; p++) {
    const page = pdfDoc.addPage([612, 792]); // US Letter
    let y = 750;

    // Title
    page.drawText('ILLINOIS DEPARTMENT OF LABOR', { x: MARGIN, y, size: 10, font: bold, color: black });
    y -= 14;
    page.drawText('CERTIFIED TRANSCRIPT OF PAYROLL', { x: MARGIN, y, size: 10, font: bold, color: black });
    y -= 20;

    // Project info
    page.drawText(`Project: ${weekRow.project.name}`, { x: MARGIN, y, size: 8, font, color: black });
    y -= 12;
    page.drawText(`Week Ending: ${weekRow.week.weekEndingDate}   Payroll #${weekRow.week.payrollNumber}   Page ${p + 1} of ${pages}`, { x: MARGIN, y, size: 8, font, color: black });
    y -= 20;

    // Column headers
    const cols = [MARGIN, 160, 250, 310, 360, 410, 460, 510];
    const headers = ['Worker / Trade', 'ST Hrs', 'OT Hrs', 'Base Rate', 'Gross', 'Deduct', 'Net Pay'];
    headers.forEach((h, i) => page.drawText(h, { x: cols[i], y, size: 7, font: bold, color: black }));
    y -= 4;
    page.drawLine({ start: { x: MARGIN, y }, end: { x: 576, y }, thickness: 0.5, color: black });
    y -= 12;

    const slice = entries.slice(p * ROWS_PER_PAGE, (p + 1) * ROWS_PER_PAGE);
    for (const e of slice) {
      const totalSt = (e.monSt + e.tueSt + e.wedSt + e.thuSt + e.friSt + e.satSt + e.sunSt);
      const totalOt = (e.monOt + e.tueOt + e.wedOt + e.thuOt + e.friOt + e.satOt + e.sunOt);
      page.drawText(e.workerName.slice(0, 25), { x: cols[0], y, size: 7, font, color: black });
      page.drawText(e.trade?.slice(0, 18) ?? '', { x: cols[0], y: y - 9, size: 6, font, color: rgb(0.4, 0.4, 0.4) });
      page.drawText(String(totalSt), { x: cols[1], y, size: 7, font, color: black });
      page.drawText(String(totalOt), { x: cols[2], y, size: 7, font, color: black });
      page.drawText(`$${e.baseRate.toFixed(2)}`, { x: cols[3], y, size: 7, font, color: black });
      page.drawText(`$${(e.grossWages ?? 0).toFixed(2)}`, { x: cols[4], y, size: 7, font, color: black });
      page.drawText(`$${(e.deductions ?? 0).toFixed(2)}`, { x: cols[5], y, size: 7, font, color: black });
      page.drawText(`$${(e.netPay ?? 0).toFixed(2)}`, { x: cols[6], y, size: 7, font, color: black });
      y -= 24;
    }

    // Certification block on last page
    if (p === pages - 1) {
      y = Math.min(y, 120);
      page.drawText('I certify under penalty of perjury that the foregoing information is true and correct.', { x: MARGIN, y, size: 7, font, color: black });
      y -= 20;
      page.drawText('Signature: _________________________________   Date: ________________', { x: MARGIN, y, size: 7, font, color: black });
    }
  }

  return pdfDoc.save();
}
```

```ts
  const ROWS_PER_PAGE = 6;
  const pages = Math.ceil(Math.max(entries.length, 1) / ROWS_PER_PAGE);

  for (let p = 0; p < pages; p++) {
    const page = pdfDoc.addPage([612, 792]);
    let y = 750;

    page.drawText('NEW YORK STATE DEPARTMENT OF LABOR', { x: 36, y, size: 9, font: bold, color: black });
    y -= 13;
    page.drawText('MONTHLY CERTIFIED PAYROLL REPORT', { x: 36, y, size: 9, font: bold, color: black });
```

### Task 5 — Certifications API route

- [ ] Create `src/server/routes/certifications.ts`:

```ts
import { Router } from 'express';
import { randomUUID } from 'crypto';
import { z } from 'zod';
import { eq } from 'drizzle-orm';
import { getDb } from '../db/index.js';
import { payrollCertifications, payrollWeeks } from '../db/schema.js';
import { requireAuth } from '../middleware/auth.js';
import { assertProjectAccess } from '../utils/assertProjectAccess.js';
import { validate } from '../middleware/validate.js';

const router = Router();
router.use(requireAuth);

const SignSchema = z.object({
  signerName: z.string().min(1).max(200),
  signerTitle: z.string().min(1).max(200),
  signatureData: z.string().min(1), // base64 PNG or "TYPED:<name>"
});

// POST /api/payroll/weeks/:weekId/certify
router.post('/:weekId/certify', validate(SignSchema), async (req, res) => {
  const { weekId } = req.params;
  const userId = req.user!.userId;
  const db = getDb();
  const [week] = await db.select({ projectId: payrollWeeks.projectId })
    .from(payrollWeeks).where(eq(payrollWeeks.id, weekId)).limit(1);
  if (!week) { res.status(404).json({ error: 'Week not found' }); return; }
  await assertProjectAccess(db, week.projectId, userId);
  const body = req.body as z.infer<typeof SignSchema>;

  // Upsert certification (one per week)
  const existing = await db.select({ id: payrollCertifications.id })
    .from(payrollCertifications).where(eq(payrollCertifications.payrollWeekId, weekId)).limit(1);

  if (existing.length) {
    await db.update(payrollCertifications).set({
      signerName: body.signerName, signerTitle: body.signerTitle,
      signedAt: new Date().toISOString(), signatureData: body.signatureData,
      signedByUserId: userId,
    }).where(eq(payrollCertifications.payrollWeekId, weekId));
  } else {
    await db.insert(payrollCertifications).values({
      id: randomUUID(), payrollWeekId: weekId, signedByUserId: userId,
      signerName: body.signerName, signerTitle: body.signerTitle,
      signedAt: new Date().toISOString(), signatureData: body.signatureData,
    });
  }
  res.json({ ok: true });
});

// GET /api/payroll/weeks/:weekId/certify
router.get('/:weekId/certify', async (req, res) => {
  const { weekId } = req.params;
  const db = getDb();
  const [cert] = await db.select().from(payrollCertifications)
    .where(eq(payrollCertifications.payrollWeekId, weekId)).limit(1);
  res.json({ certification: cert ?? null });
});

export default router;
```

```tsx
import { useRef, useState } from 'react';

interface Props { onSave: (dataUrl: string) => void; onCancel: () => void; }

export function SignaturePad({ onSave, onCancel }: Props) {
  const canvasRef = useRef<HTMLCanvasElement>(null);
  const [drawing, setDrawing] = useState(false);
  const [typedName, setTypedName] = useState('');
  const [mode, setMode] = useState<'draw' | 'type'>('type');

  function getPos(e: React.MouseEvent<HTMLCanvasElement>) {
    const rect = canvasRef.current!.getBoundingClientRect();
    return { x: e.clientX - rect.left, y: e.clientY - rect.top };
  }

  function onMouseDown(e: React.MouseEvent<HTMLCanvasElement>) {
    setDrawing(true);
    const ctx = canvasRef.current!.getContext('2d')!;
    const { x, y } = getPos(e);
    ctx.beginPath(); ctx.moveTo(x, y);
  }
  function onMouseMove(e: React.MouseEvent<HTMLCanvasElement>) {
    if (!drawing) return;
    const ctx = canvasRef.current!.getContext('2d')!;
    ctx.lineWidth = 2; ctx.strokeStyle = '#1a1a1a'; ctx.lineCap = 'round';
    const { x, y } = getPos(e);
    ctx.lineTo(x, y); ctx.stroke();
  }
  function clear() {
    const ctx = canvasRef.current?.getContext('2d');
    if (ctx) ctx.clearRect(0, 0, canvasRef.current!.width, canvasRef.current!.height);
  }
  function save() {
    if (mode === 'type') { onSave(`TYPED:${typedName}`); return; }
    onSave(canvasRef.current!.toDataURL('image/png'));
  }

  return (
    <div className="space-y-4">
      <div className="flex gap-4 text-sm">
        <button onClick={() => setMode('type')} className={mode === 'type' ? 'font-semibold underline' : 'text-gray-500'}>Type name</button>
        <button onClick={() => setMode('draw')} className={mode === 'draw' ? 'font-semibold underline' : 'text-gray-500'}>Draw signature</button>
      </div>
      {mode === 'type' && (
        <input value={typedName} onChange={e => setTypedName(e.target.value)}
          placeholder="Full legal name" className="w-full border rounded px-3 py-2 text-sm font-serif italic" />
      )}
      {mode === 'draw' && (
        <div>
          <canvas ref={canvasRef} width={400} height={120}
            className="border border-gray-300 rounded cursor-crosshair w-full"
            onMouseDown={onMouseDown} onMouseMove={onMouseMove}
            onMouseUp={() => setDrawing(false)} onMouseLeave={() => setDrawing(false)} />
          <button onClick={clear} className="text-xs text-gray-500 mt-1">Clear</button>
        </div>
      )}
      <div className="flex gap-3">
        <button onClick={save} disabled={mode === 'type' && !typedName.trim()}
          className="bg-yellow-700 text-white px-4 py-2 rounded text-sm disabled:opacity-50">
          Apply Signature
        </button>
        <button onClick={onCancel} className="text-sm text-gray-500">Cancel</button>
      </div>
    </div>
  );
}
```

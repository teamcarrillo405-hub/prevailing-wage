### Task 1 — Research Gusto export column signature

Gusto's "Payroll Journal" export (CSV) has this header signature:


- [ ] Create `src/server/services/import/gustoMapper.ts`:

```ts
import type { MappedEntry, ImportProvider } from '../importService.js';

// Gusto "Payroll Journal" CSV column signature
const GUSTO_COLUMNS = ['Employee Name', 'Employee ID', 'Earnings Type', 'Hours', 'Rate', 'Amount'];

export function isGustoFormat(headers: string[]): boolean {
  return GUSTO_COLUMNS.every(col => headers.some(h => h.trim().toLowerCase() === col.toLowerCase()));
}

export function mapGustoRows(rows: Record<string, string>[]): MappedEntry[] {
  // Group by employee — Gusto has one row per earnings type per employee
  const byEmployee = new Map<string, { name: string; stHours: number; otHours: number; rate: number; gross: number }>();

  for (const row of rows) {
    const name = row['Employee Name']?.trim();
    if (!name) continue;
    const earningsType = (row['Earnings Type'] ?? '').toLowerCase();
    const hours = parseFloat(row['Hours'] ?? '0') || 0;
    const rate = parseFloat(row['Rate'] ?? '0') || 0;
    const amount = parseFloat(row['Amount'] ?? '0') || 0;

    if (!byEmployee.has(name)) byEmployee.set(name, { name, stHours: 0, otHours: 0, rate, gross: 0 });
    const emp = byEmployee.get(name)!;
    emp.gross += amount;
    if (earningsType.includes('overtime') || earningsType.includes('ot')) {
      emp.otHours += hours;
    } else if (earningsType.includes('regular') || earningsType.includes('straight')) {
      emp.stHours += hours;
      if (rate > 0) emp.rate = rate;
    }
  }

  return Array.from(byEmployee.values()).map(emp => ({
    workerNameCsv: emp.name,
    workerId: null,        // matched in importService
    classificationId: null,
    baseRateSnapshot: emp.rate,
    fringeRateSnapshot: 0,
    grossWages: emp.gross,
    // Distribute ST hours across Mon-Fri evenly for preview
    monSt: Math.floor(emp.stHours / 5), tueSt: Math.floor(emp.stHours / 5),
    wedSt: Math.floor(emp.stHours / 5), thuSt: Math.floor(emp.stHours / 5),
    friSt: emp.stHours - Math.floor(emp.stHours / 5) * 4,
    satSt: 0, sunSt: 0,
    monOt: Math.floor(emp.otHours / 5), tueOt: Math.floor(emp.otHours / 5),
    wedOt: Math.floor(emp.otHours / 5), thuOt: Math.floor(emp.otHours / 5),
    friOt: emp.otHours - Math.floor(emp.otHours / 5) * 4,
    satOt: 0, sunOt: 0,
    monDt: 0, tueDt: 0, wedDt: 0, thuDt: 0, friDt: 0, satDt: 0, sunDt: 0,
    deductions: null, netPay: null,
  }));
}

export const gustoProvider: ImportProvider = {
  id: 'gusto',
  label: 'Gusto',
  detect: isGustoFormat,
  map: mapGustoRows,
};
```

### Task 2 — Paychex mapper

Paychex "Preview Payroll" export signature:
```
Name, File Number, Department, Job, Check Date, Regular Hours, OT Hours, Regular Earnings, OT Earnings, Total Earnings, Total Deductions, Net Pay
```

- [ ] Create `src/server/services/import/paychexMapper.ts`:

```ts
import type { MappedEntry, ImportProvider } from '../importService.js';

const PAYCHEX_COLUMNS = ['File Number', 'Regular Hours', 'OT Hours', 'Regular Earnings', 'Net Pay'];

export function isPaychexFormat(headers: string[]): boolean {
  return PAYCHEX_COLUMNS.every(col => headers.some(h => h.trim().toLowerCase() === col.toLowerCase()));
}

export function mapPaychexRows(rows: Record<string, string>[]): MappedEntry[] {
  return rows
    .filter(r => r['Name']?.trim())
    .map(row => {
      const stHours = parseFloat(row['Regular Hours'] ?? '0') || 0;
      const otHours = parseFloat(row['OT Hours'] ?? '0') || 0;
      const grossWages = parseFloat(row['Total Earnings'] ?? '0') || 0;
      const deductions = parseFloat(row['Total Deductions'] ?? '0') || 0;
      const netPay = parseFloat(row['Net Pay'] ?? '0') || 0;
      const baseRate = stHours > 0 ? parseFloat(row['Regular Earnings'] ?? '0') / stHours : 0;

      return {
        workerNameCsv: row['Name'].trim(),
        workerId: null, classificationId: null,
        baseRateSnapshot: Math.round(baseRate * 100) / 100,
        fringeRateSnapshot: 0,
        grossWages, deductions, netPay,
        monSt: Math.floor(stHours / 5), tueSt: Math.floor(stHours / 5),
        wedSt: Math.floor(stHours / 5), thuSt: Math.floor(stHours / 5),
        friSt: stHours - Math.floor(stHours / 5) * 4,
        satSt: 0, sunSt: 0,
        monOt: Math.floor(otHours / 5), tueOt: Math.floor(otHours / 5),
        wedOt: Math.floor(otHours / 5), thuOt: Math.floor(otHours / 5),
        friOt: otHours - Math.floor(otHours / 5) * 4,
        satOt: 0, sunOt: 0,
        monDt: 0, tueDt: 0, wedDt: 0, thuDt: 0, friDt: 0, satDt: 0, sunDt: 0,
      } satisfies MappedEntry;
    });
}

export const paychexProvider: ImportProvider = {
  id: 'paychex',
  label: 'Paychex',
  detect: isPaychexFormat,
  map: mapPaychexRows,
};
```

### Task 8 — SAM.gov opportunities lookup route

- [ ] Create `src/server/routes/samLookup.ts`:

```ts
import { Router } from 'express';
import { requireAuth } from '../middleware/auth.js';

const router = Router();
router.use(requireAuth);

const SAM_API_KEY = process.env.SAM_API_KEY ?? '';
const SAM_BASE = 'https://api.sam.gov/opportunities/v2/search';

interface SamOpportunity {
  noticeId: string; title: string; solicitationNumber: string;
  organizationName: string; organizationId: string;
  postedDate: string; responseDeadLine: string | null;
  naicsCode: string | null; placeOfPerformance: { state: { code: string } | null } | null;
}

// GET /api/sam-lookup?contract=<solicitation_number>
router.get('/', async (req, res) => {
  const contract = (req.query.contract as string)?.trim();
  if (!contract) { res.status(400).json({ error: 'contract query param required' }); return; }

  const url = new URL(SAM_BASE);
  url.searchParams.set('api_key', SAM_API_KEY);
  url.searchParams.set('solicitationNumber', contract);
  url.searchParams.set('limit', '5');

  try {
    const resp = await fetch(url.toString());
    if (!resp.ok) { res.status(502).json({ error: 'SAM.gov API error' }); return; }
    const data = await resp.json() as { opportunitiesData?: SamOpportunity[]; totalRecords?: number };
    const opps = data.opportunitiesData ?? [];
    res.json({
      results: opps.map(o => ({
        noticeId: o.noticeId,
        title: o.title,
        solicitationNumber: o.solicitationNumber,
        awardingAgency: o.organizationName,
        state: o.placeOfPerformance?.state?.code ?? null,
        postedDate: o.postedDate,
      })),
    });
  } catch (err) {
    console.error('[samLookup] fetch error:', err);
    res.status(502).json({ error: 'SAM.gov request failed' });
  }
});

export default router;
```

### Task 9 — Project create form: contract number lookup

- [ ] In the project creation form (find in `ProjectDetailPage.tsx` or `ProjectForm` component), add a contract number field with a "Look up" button:

```tsx
const [contractQuery, setContractQuery] = useState('');
const [samResults, setSamResults] = useState<SamResult[]>([]);
const [samLoading, setSamLoading] = useState(false);

async function lookupContract() {
  setSamLoading(true);
  const r = await fetch(`/api/sam-lookup?contract=${encodeURIComponent(contractQuery)}`, { credentials: 'include' });
  const data = await r.json();
  setSamResults(data.results ?? []);
  setSamLoading(false);
}

function applyResult(r: SamResult) {
  // Pre-fill project name, awarding agency fields
  setValue('name', r.title);
  setValue('awardingAgency', r.awardingAgency);
  setValue('contractNumber', r.solicitationNumber);
  if (r.state) setValue('state', r.state);
  setSamResults([]);
}
```

```tsx
<div className="space-y-2">
  <label className="block text-xs font-medium text-gray-700">Contract / Solicitation Number (optional)</label>
  <div className="flex gap-2">

      placeholder="e.g. W912BU-25-R-0001"
      className="flex-1 border border-gray-300 rounded px-3 py-2 text-sm" />
    <button type="button" onClick={lookupContract} disabled={!contractQuery || samLoading}

</div>
```

# v4.1 — Stripe Billing + Alerts + Audit Package Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add Stripe subscription billing (Starter free / Pro $49/mo), WD change-detection alerts and violation alert emails via resend, and a one-click audit package ZIP download containing all WH-347 PDFs + compliance summary for a project.

**Architecture:** Stripe Checkout for billing (redirect flow, no embedded card UI needed). WD change detection runs in the existing `node-cron` scheduler — compare `lastModified` from SAM.gov WDOL against cached value, email owner if changed. Audit package uses `archiver` (new dep) to stream a ZIP of on-the-fly-generated PDFs.

**Tech Stack:** `stripe` (new), `archiver` (new), resend (existing), node-cron (existing), pdf-lib (existing).

**Install before starting:**
```bash
npm install stripe archiver
npm install -D @types/archiver
```

```ts
let _stripe: any = null;
function getStripe() {
  if (!process.env.STRIPE_SECRET_KEY) return null;
  if (!_stripe) {
    const Stripe = require('stripe');
    _stripe = new Stripe(process.env.STRIPE_SECRET_KEY, { apiVersion: '2024-06-20' });
  }
  return _stripe;
}

export const PLANS = {
  pro: { priceId: process.env.STRIPE_PRO_PRICE_ID ?? '', label: 'Pro', maxMembers: 10 },
} as const;

export async function createCheckoutSession(opts: {
  userId: string; email: string; priceId: string; successUrl: string; cancelUrl: string;
}): Promise<string | null> {
  const stripe = getStripe();
  if (!stripe) return null;
  const session = await stripe.checkout.sessions.create({
    mode: 'subscription',
    customer_email: opts.email,
    line_items: [{ price: opts.priceId, quantity: 1 }],
    success_url: opts.successUrl,
    cancel_url: opts.cancelUrl,
    metadata: { userId: opts.userId },
  });
  return session.url as string;
}

export async function createPortalSession(customerId: string, returnUrl: string): Promise<string | null> {
  const stripe = getStripe();
  if (!stripe) return null;
  const session = await stripe.billingPortal.sessions.create({ customer: customerId, return_url: returnUrl });
  return session.url as string;
}

export function constructWebhookEvent(rawBody: Buffer, sig: string): any {
  const stripe = getStripe();
  if (!stripe) throw new Error('Stripe not configured');
  return stripe.webhooks.constructEvent(rawBody, sig, process.env.STRIPE_WEBHOOK_SECRET!);
}
```

```ts
import { Router, raw } from 'express';
import { requireAuth } from '../middleware/auth.js';
import { getDb } from '../db/index.js';
import { users } from '../db/schema.js';
import { eq } from 'drizzle-orm';
import { createCheckoutSession, createPortalSession, constructWebhookEvent, PLANS } from '../services/stripeService.js';

const router = Router();
const APP_URL = process.env.APP_URL || 'http://localhost:3000';

// GET /api/billing/status — current plan for logged-in user
router.get('/status', requireAuth, async (req, res) => {
  const db = getDb();
  const [user] = await db.select({
    planTier: users.planTier,
    stripeSubscriptionStatus: users.stripeSubscriptionStatus,
  }).from(users).where(eq(users.id, req.user!.userId)).limit(1);
  res.json({ planTier: user?.planTier ?? 'starter', subscriptionStatus: user?.stripeSubscriptionStatus });
});

// POST /api/billing/checkout — create Stripe Checkout session
router.post('/checkout', requireAuth, async (req, res) => {
  const db = getDb();
  const [user] = await db.select({ email: users.email }).from(users).where(eq(users.id, req.user!.userId)).limit(1);
  const url = await createCheckoutSession({
    userId: req.user!.userId,
    email: user!.email,
    priceId: PLANS.pro.priceId,
    successUrl: `${APP_URL}/billing?success=1`,
    cancelUrl: `${APP_URL}/billing?cancel=1`,
  });
  if (!url) { res.status(503).json({ error: 'Billing not configured' }); return; }
  res.json({ url });
});

// POST /api/billing/portal — Stripe customer portal for manage/cancel
router.post('/portal', requireAuth, async (req, res) => {
  const db = getDb();
  const [user] = await db.select({ stripeCustomerId: users.stripeCustomerId }).from(users).where(eq(users.id, req.user!.userId)).limit(1);
  if (!user?.stripeCustomerId) { res.status(400).json({ error: 'No billing account found' }); return; }
  const url = await createPortalSession(user.stripeCustomerId, `${APP_URL}/billing`);
  res.json({ url });
});

// POST /api/billing/webhook — Stripe events (raw body required)
router.post('/webhook', raw({ type: 'application/json' }), async (req, res) => {
  const sig = req.headers['stripe-signature'] as string;
  let event: any;
  try { event = constructWebhookEvent(req.body, sig); }
  catch (err: any) { res.status(400).send(`Webhook error: ${err.message}`); return; }

  const db = getDb();
  if (event.type === 'checkout.session.completed') {
    const session = event.data.object;
    const userId = session.metadata?.userId;
    if (userId) {
      await db.update(users).set({
        stripeCustomerId: session.customer,
        stripeSubscriptionId: session.subscription,
        stripeSubscriptionStatus: 'active',
        planTier: 'pro',
      }).where(eq(users.id, userId));
    }
  }
  if (event.type === 'customer.subscription.deleted' || event.type === 'customer.subscription.updated') {
    const sub = event.data.object;
    const status = sub.status;
    const isActive = status === 'active' || status === 'trialing';
    await db.update(users).set({
      stripeSubscriptionStatus: status,
      planTier: isActive ? 'pro' : 'starter',
    }).where(eq(users.stripeSubscriptionId, sub.id));
  }

  res.json({ received: true });
});

export default router;
```

### Task 5 — wdChangeDetector service

- [ ] Create `src/server/services/wdChangeDetector.ts`:

```ts
import { getDb } from '../db/index.js';
import { wdCache, projectMembers, users, projects } from '../db/schema.js';
import { eq, isNull } from 'drizzle-orm';
import { sendWdChangedEmail } from './emailService.js';

const WDOL_BASE = 'https://api.sam.gov/wage-determinations/v2';

export async function checkWdChanges(): Promise<void> {
  const db = getDb();
  // Get all distinct WD numbers referenced by active projects
  const rows = await db
    .select({ wdNumber: wdCache.wdNumber, lastModified: wdCache.lastCheckedAt, projectId: projects.id })
    .from(projects)
    .innerJoin(wdCache, eq(wdCache.wdNumber, projects.wageDecisionNumber))
    .where(isNull(projects.archivedAt))
    .limit(100);

  for (const row of rows) {
    try {
      const resp = await fetch(`${WDOL_BASE}/${row.wdNumber}`, { headers: { Accept: 'application/json' } });
      if (!resp.ok) continue;
      const data = await resp.json() as any;
      const remoteRevision = data?.wageDecision?.revisionNumber as string | undefined;
      const [cached] = await db.select({ revision: wdCache.revisionNumber })
        .from(wdCache).where(eq(wdCache.wdNumber, row.wdNumber)).limit(1);
      if (remoteRevision && cached?.revision && remoteRevision !== cached.revision) {
        await sendWdChangedEmail({ projectId: row.projectId, wdNumber: row.wdNumber, oldRev: cached.revision, newRev: remoteRevision });
        await db.update(wdCache).set({ revisionNumber: remoteRevision, lastCheckedAt: new Date().toISOString() })
          .where(eq(wdCache.wdNumber, row.wdNumber));
      }
    } catch (err) {
      console.error('[wdChangeDetector] error checking', row.wdNumber, err);
    }
  }
}
```

```ts
export async function sendViolationAlertEmail(opts: {
  projectId: string; weekId: string;
  violations: Array<{ workerName: string; violationType: string; delta: number }>;
}) {
  const resend = await getResend();
  if (!resend) return;
  const memberRows = await getProjectMemberRows(opts.projectId);
  const owners = memberRows.filter(r => r.role === 'owner' && r.email);
  const summary = opts.violations.slice(0, 5)
    .map(v => `<li>${v.workerName}: ${v.violationType} (Δ $${Math.abs(v.delta).toFixed(2)})</li>`).join('');
  for (const owner of owners) {
    try {
      await resend.emails.send({
        from: FROM_EMAIL, to: owner.email!,
        subject: `Compliance violations detected — ${opts.violations.length} issue(s)`,
        html: `<p>${opts.violations.length} payroll compliance violation(s) detected:</p><ul>${summary}</ul><a href="${APP_URL}/projects/${opts.projectId}/payroll/${opts.weekId}">Review payroll week →</a>`,
      });
    } catch (err) { console.error('[emailService] sendViolationAlertEmail failed:', err); }
  }
}
```

```ts
import { Router } from 'express';
import archiver from 'archiver';
import { requireAuth } from '../middleware/auth.js';
import { assertProjectAccess } from '../utils/assertProjectAccess.js';
import { getDb } from '../db/index.js';
import { payrollWeeks } from '../db/schema.js';
import { eq, isNull } from 'drizzle-orm';
import { generateWh347 } from '../services/wh347Generator.js';
import { computeCompliance } from '../services/complianceService.js';

const router = Router();

// GET /api/projects/:id/audit-package — streams ZIP of all WH-347s + compliance summary
router.get('/:id/audit-package', requireAuth, async (req, res) => {
  const projectId = req.params.id as string;
  const userId = req.user!.userId;
  const db = getDb();

  try { await assertProjectAccess(db, projectId, userId); }
  catch (err: any) { res.status(err.status ?? 500).json({ error: err.message }); return; }

  const weeks = await db.select().from(payrollWeeks)
    .where(eq(payrollWeeks.projectId, projectId))
    .orderBy(payrollWeeks.payrollNumber);

  res.setHeader('Content-Type', 'application/zip');
  res.setHeader('Content-Disposition', `attachment; filename="audit-package-${projectId}.zip"`);

  const archive = archiver('zip', { zlib: { level: 6 } });
  archive.pipe(res);

  // Add WH-347 PDF for each submitted week
  for (const week of weeks) {
    try {
      const pdfBytes = await generateWh347(week.id);
      archive.append(Buffer.from(pdfBytes), { name: `wh347-week-${week.payrollNumber}.pdf` });
    } catch (err) {
      console.error('[auditExport] failed generating WH-347 for week', week.id, err);
    }
  }

  // Add compliance summary CSV
  const rows = ['Worker,Week,Violation Type,Expected,Actual,Delta'];
  for (const week of weeks) {
    try {
      const result = await computeCompliance(week.id);
      for (const v of result.violations) {
        rows.push(`"${v.workerName}",${week.weekEndingDate},"${v.violationType}",${v.expected},${v.actual},${v.delta}`);
      }
    } catch { /* skip */ }
  }
  archive.append(rows.join('\n'), { name: 'compliance-summary.csv' });

  await archive.finalize();
});

export default router;
```
